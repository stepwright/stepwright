reminder.cpp
struct reminder : stepwright::machine<reminder> {
    stepwright::delay pause;
    explicit reminder(stepwright::timer_service& t)
        : pause(t, 25ms) {}
    step begin() {
        return after<&reminder::announce>(pause);
    }
    step announce() {
        std::puts("ready");
        return done();
    }
};
