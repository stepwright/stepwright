cmake_minimum_required(VERSION 3.21)
project(stepwright_consumer LANGUAGES CXX)

set(STEPWRIGHT_CONSUMER_MODE "install" CACHE STRING "install|fetch|subdirectory")
if(STEPWRIGHT_CONSUMER_MODE STREQUAL "install")
    find_package(stepwright 0.2 CONFIG QUIET)
    if(stepwright_FOUND)
        message(FATAL_ERROR "A 0.1 package incorrectly accepted a 0.2 request")
    endif()
    find_package(stepwright 0.1 CONFIG REQUIRED)
elseif(STEPWRIGHT_CONSUMER_MODE STREQUAL "fetch")
    include(FetchContent)
    FetchContent_Declare(stepwright SOURCE_DIR "${STEPWRIGHT_SOURCE}")
    FetchContent_MakeAvailable(stepwright)
elseif(STEPWRIGHT_CONSUMER_MODE STREQUAL "subdirectory")
    add_subdirectory("${STEPWRIGHT_SOURCE}" stepwright)
else()
    message(FATAL_ERROR "Unknown consumer mode: ${STEPWRIGHT_CONSUMER_MODE}")
endif()

foreach(unwanted test_smoke all_examples all_benchmarks)
    if(TARGET ${unwanted})
        message(FATAL_ERROR "stepwright leaked developer targets: ${unwanted}")
    endif()
endforeach()
add_executable(consumer main.cpp peer.cpp)
target_link_libraries(consumer PRIVATE stepwright::stepwright)
enable_testing()
add_test(NAME consumer COMMAND consumer)

# Every installed/public header must be self-contained, not just the umbrella.
get_target_property(stepwright_include stepwright::stepwright INTERFACE_INCLUDE_DIRECTORIES)
if(STEPWRIGHT_CONSUMER_MODE STREQUAL "install")
    set(header_root "${stepwright_include}")
else()
    set(header_root "${STEPWRIGHT_SOURCE}/include")
endif()
file(GLOB_RECURSE headers CONFIGURE_DEPENDS "${header_root}/stepwright/*.hpp")
foreach(header IN LISTS headers)
    file(RELATIVE_PATH name "${header_root}" "${header}")
    string(MAKE_C_IDENTIFIER "${name}" id)
    file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/${id}.cpp" "#include <${name}>\n")
    add_library(header_${id} OBJECT "${CMAKE_CURRENT_BINARY_DIR}/${id}.cpp")
    target_link_libraries(header_${id} PRIVATE stepwright::stepwright)
endforeach()
